import {RootScene, IScene, ISceneEventMap, lerp} from 'threepipe'
import {Scene} from 'three'

declare const symbolEvent: unique symbol
interface CustomEvents extends ISceneEventMap {
    custom: {value: number}
    sceneUpdate: ISceneEventMap['sceneUpdate'] & {customToken: string}
    42: {value: number}
    [symbolEvent]: {value: number}
}

declare const scene: RootScene<CustomEvents>
const asInterface: IScene<CustomEvents> = scene
const asBase: Scene<CustomEvents> = scene
void asInterface
void asBase

function registerRefresh<TE extends ISceneEventMap>(genericScene: RootScene<TE>) {
    genericScene.addEventListener('objectUpdate', genericScene.refreshScene)
    genericScene.addEventListener('geometryUpdate', genericScene.refreshScene)
    genericScene.addEventListener('geometryChanged', genericScene.refreshScene)
}
void registerRefresh

scene.addEventListener('custom', event => {
    const value: number = event.value
    const type: 'custom' = event.type
    const target: RootScene<CustomEvents> = event.target
    void value
    void type
    void target
    // @ts-expect-error Custom event data must retain its declared type.
    const wrong: string = event.value
    void wrong
})
scene.addEventListener('sceneUpdate', event => {
    const token: string = event.customToken
    void token
})
// @ts-expect-error Custom payload refinements must not accept an incompatible listener.
scene.addEventListener('sceneUpdate', (event: {customToken: number}) => { void event })
// @ts-expect-error Unknown event names are rejected.
scene.addEventListener('unknown', () => undefined)
// @ts-expect-error Three.js event names must be strings.
scene.addEventListener(42, () => undefined)
// @ts-expect-error Three.js event names must be strings.
scene.addEventListener(symbolEvent, () => undefined)
